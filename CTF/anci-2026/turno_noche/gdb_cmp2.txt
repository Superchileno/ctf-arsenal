import gdb, re
s = gdb.execute("info files", to_string=True)
m = re.search(r"Entry point: (0x[0-9a-fA-F]+)", s)
print("ENTRY", m.group(1) if m else "none")
base = int(m.group(1), 16) - 0x12b0
print("BASE", hex(base))
class CmpBP(gdb.Breakpoint):
    def stop(self):
        rax = int(gdb.parse_and_eval("$rax"))
        r10 = int(gdb.parse_and_eval("$r10"))
        rdx = int(gdb.parse_and_eval("$rdx"))
        ip = int(gdb.parse_and_eval("*(unsigned short*)($rax+0x1010)"))
        a = int(gdb.parse_and_eval("*(unsigned short*)($rax+$r10*2+0x1000)"))
        b = int(gdb.parse_and_eval("*(unsigned short*)($rax+$rdx*2+0x1000)"))
        print("CMP a=%d b=%d ip=%d" % (a, b, ip))
        return False
